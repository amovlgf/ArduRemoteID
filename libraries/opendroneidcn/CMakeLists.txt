cmake_minimum_required(VERSION 3.10)
project(opendroneidcn C)
option(CN46750_SANITIZE "Run host tests with address and undefined-behavior sanitizers" OFF)
add_library(opendroneidcn STATIC src/opendroneidcn.c)
target_include_directories(opendroneidcn PUBLIC src)
target_compile_features(opendroneidcn PUBLIC c_std_99)
target_compile_options(opendroneidcn PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(opendroneidcn PUBLIC m)
add_executable(cn46750_tests tests/test_cn46750.c)
target_compile_options(cn46750_tests PRIVATE -Wall -Wextra -Werror -pedantic)
target_compile_definitions(cn46750_tests PRIVATE CN_VECTOR_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/vectors")
target_link_libraries(cn46750_tests PRIVATE opendroneidcn)
if(CN46750_SANITIZE)
  foreach(target opendroneidcn cn46750_tests)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
  endforeach()
  target_link_libraries(cn46750_tests PRIVATE -fsanitize=address,undefined,float-cast-overflow)
endif()
enable_testing()
add_test(NAME cn46750 COMMAND cn46750_tests)
